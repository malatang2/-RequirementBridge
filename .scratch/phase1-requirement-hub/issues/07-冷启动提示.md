```typescript
import type { ApiDraft, RequirementDraft } from "@/types/database";

/** 分组结果：有归属（按需求分组）+ 未归属组 */
export interface ApiDraftGroup {
  requirement: { id: string; title: string; lifecycle: string; priority: string } | null;
  drafts: ApiDraft[];
}

/**
 * 把 API 草稿按 source_requirement_id 分组。
 * - 有 source_requirement_id 且能在 requirements 里找到对应需求的 → 归到该需求组
 * - 没关联（null）或关联的需求不在 requirements 列表里（被删/跨项目）→ 归"未归属"组
 * 纯函数，不查 DB。requirements 由调用方先查好传入。
 */
export function groupApiDraftsByRequirement(
  drafts: ApiDraft[],
  requirements: RequirementDraft[]
): ApiDraftGroup[] {
  const reqMap = new Map(requirements.map(r => [r.id, r]));
  const grouped = new Map<string, ApiDraft[]>();
  const unattached: ApiDraft[] = [];

  for (const d of drafts) {
    const rid = d.source_requirement_id;
    if (rid && reqMap.has(rid)) {
      const arr = grouped.get(rid) ?? [];
      arr.push(d);
      grouped.set(rid, arr);
    } else {
      unattached.push(d);
    }
  }

  const groups: ApiDraftGroup[] = Array.from(grouped.entries()).map(([rid, ds]) => {
    const r = reqMap.get(rid)!;
    return {
      requirement: { id: r.id, title: r.title, lifecycle: r.lifecycle, priority: r.priority },
      drafts: ds,
    };
  });
  // 未归属组放最后
  if (unattached.length > 0) {
    groups.push({ requirement: null, drafts: unattached });
  }
  return groups;
}
```

**关键设计点**：
- 纯函数不查 DB——`requirements` 由调用方（页面 server component）批量查好传入，避免 N+1
- 关联需求"找不到"的情况（source_requirement_id 非空但 requirements 里没有——比如需求被软删了）归入"未归属"组，**不报错**（防御性）
- 分组顺序：建议按需求 lifecycle asc → 优先级 desc（复用 03 的 LIFECYCLE_ORDER 排序思路），未归属组恒在最后
- 组内 drafts 顺序保持传入顺序（页面查询时已 order by created_at desc）
